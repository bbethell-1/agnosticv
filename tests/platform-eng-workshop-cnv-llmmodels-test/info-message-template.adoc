=== Usage Examples

ifdef::litellm_model_granite_3_2_8b_instruct[]
==== Granite 3.2 8B Instruct (Chat)

[source,bash,subs="attributes"]
----
curl {litellm_api_base_url}/chat/completions \
  -H "Authorization: Bearer {litellm_virtual_key}" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "granite-3-2-8b-instruct",
    "messages": [{"role": "user", "content": "Explain Kubernetes"}]
  }'
----

==== Python Example

[source,python,subs="attributes"]
----
from openai import OpenAI

client = OpenAI(
    api_key="{litellm_virtual_key}",
    base_url="{litellm_api_base_url}"
)

ifdef::litellm_model_granite_3_2_8b_instruct,litellm_model_mistral_7b_instruct,litellm_model_granite_8b_code_instruct[]
# Chat completion
response = client.chat.completions.create(
    model="granite-3-2-8b-instruct",  # or mistral-7b-instruct or granite-8b-code-instruct-128k
    messages=[{{"role": "user", "content": "What is Red Hat OpenShift?"}}]
)
print(response.choices[0].message.content)

endif::[]
ifdef::litellm_model_codellama_7b[]
# Code completion
response = client.completions.create(
    model="codellama-7b",
    prompt="def fibonacci(n):\n    ",
    max_tokens=100
)
print(response.choices[0].text)

endif::[]
ifdef::litellm_model_nomic_embed[]
# Embeddings
embedding = client.embeddings.create(
    model="nomic-embed-text-v1-5",
    input="Text to embed"
)
print(f"Embedding dimension: {{len(embedding.data[0].embedding)}}")
endif::[]
----
